async function loadStationSchedule(lineCode, stationCode, elementId) { let isTransfer = TRANSFER_STATIONS[stationCode] !== undefined; let linesToFetch = isTransfer ? TRANSFER_STATIONS[stationCode] : [lineCode]; let allTrains = []; let now = new Date(); for (let lCode of linesToFetch) { try { let res = await fetch(`https://rt.data.gov.hk/v1/transport/mtr/getSchedule.php?line=${lCode}&sta=${stationCode}`); let json = await res.json(); let key = `${lCode}-${stationCode}`; if (json && json.status === 1 && json.data && json.data[key]) { let sData = json.data[key]; let rawList = (sData.UP || []).concat(sData.DOWN || []); for (let item of rawList) { if (item.time) { let tTime = new Date(item.time.replace(/-/g, "/")); // 【關鍵修改】計算精確的剩餘秒數 let diffDiffSeconds = Math.floor((tTime - now) / 1000); // 只要剩餘秒數大於 -20 秒（考慮過站緩衝），都納入顯示 if (diffDiffSeconds >= -20) { // 計算分鐘數（無條件捨去，讓剩餘 59 秒內直接歸零） let diffMins = Math.floor(diffDiffSeconds / 60); allTrains.push({ lineCode: lCode, rawTime: tTime, countdown: diffMins < 0 ? 0 : diffMins, // 小於 0 均視為 0 分鐘（即將到達） destination: translateDest(item.dest) }); } } } } } catch (e) { console.error(e); } } allTrains.sort((a, b) => a.rawTime - b.rawTime); let el = document.getElementById(elementId); if (allTrains.length > 0) { let html = ""; let displayLimit = isTransfer ? 6 : 3; for (let i = 0; i < Math.min(allTrains.length, displayLimit); i++) { let t = allTrains[i]; let lObj = MTR_LINES[t.lineCode]; // 【關鍵修改】0 分鐘或更少時，直接提示「即將到達」 let timeStr = t.countdown <= 0 ? "即將到達" : `${t.countdown} 分鐘`; let timeClass = t.countdown <= 0 ? "time-soon" : "time-normal"; html += `
${lObj ? lObj.name : t.lineCode}
${timeStr}
➔ ${t.destination}
`; } el.innerHTML = html; } else { el.innerHTML = `
目前尚無即時班次或收車中
`; } }
